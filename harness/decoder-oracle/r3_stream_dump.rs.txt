// ---- helpers (session/tests.rs) ----
fn flight_stream<'a>(
    schema: &Schema,
    batches: impl IntoIterator<Item = &'a RecordBatch>,
) -> Vec<FlightData> {
    flight_stream_with(IpcWriteOptions::default(), schema, batches)
}

fn flight_stream_with<'a>(
    options: IpcWriteOptions,
    schema: &Schema,
    batches: impl IntoIterator<Item = &'a RecordBatch>,
) -> Vec<FlightData> {
    let generator = IpcDataGenerator::default();
    let mut tracker = DictionaryTracker::new(false);
    let mut context = IpcWriteContext::default();
    let mut data = vec![
        generator
            .schema_to_bytes_with_dictionary_tracker(schema, &mut tracker, &options)
            .into(),
    ];
    for batch in batches {
        let (dictionaries, encoded) = generator
            .encode(batch, &mut tracker, &options, &mut context)
            .unwrap();
        data.extend(dictionaries.into_iter().map(FlightData::from));
        data.push(encoded.into());
    }
    data
}

fn declared_as(declared: &Schema, batch: &RecordBatch) -> Vec<FlightData> {
    let mut data = flight_stream(&batch.schema(), [batch]);
    let mut tracker = DictionaryTracker::new(false);
    data[0] = IpcDataGenerator::default()
        .schema_to_bytes_with_dictionary_tracker(
            declared,
            &mut tracker,
            &IpcWriteOptions::default(),
        )
        .into();
    data
}

#[derive(Clone, Copy)]
enum Slot {
    NodeLength(usize),
    NodeNulls(usize),
    BufferOffset(usize),
    BufferLength(usize),
}

/// Rewrite one i64 of a node or buffer entry in place. FieldNode and Buffer
/// are 16-byte flatbuffer structs stored inline, so the header stays valid.
fn patch(data: &mut FlightData, slot: Slot, value: i64) {
    let mut header = data.data_header.to_vec();
    let position = {
        let message = arrow_ipc::root_as_message(&header).unwrap();
        let batch = message
            .header_as_record_batch()
            .or_else(|| message.header_as_dictionary_batch().and_then(|d| d.data()))
            .unwrap();
        let (bytes, index, half) = match slot {
            Slot::NodeLength(i) => (batch.nodes().unwrap().bytes(), i, 0),
            Slot::NodeNulls(i) => (batch.nodes().unwrap().bytes(), i, 1),
            Slot::BufferOffset(i) => (batch.buffers().unwrap().bytes(), i, 0),
            Slot::BufferLength(i) => (batch.buffers().unwrap().bytes(), i, 1),
        };
        bytes.as_ptr() as usize - header.as_ptr() as usize + index * 16 + half * 8
    };
    header[position..position + 8].copy_from_slice(&value.to_le_bytes());
    data.data_header = Bytes::from(header);
}

    fn dump(name: &str, data: &[FlightData]) {
        let Ok(dir) = std::env::var("R3_STREAMS") else { return };
        let mut out = Vec::new();
        for m in data {
            let mut h = m.data_header.to_vec();
            while h.len() % 8 != 0 {
                h.push(0);
            }
            out.extend(0xFFFF_FFFFu32.to_le_bytes());
            out.extend((h.len() as i32).to_le_bytes());
            out.extend(&h);
            out.extend(&m.data_body[..]);
        }
        out.extend(0xFFFF_FFFFu32.to_le_bytes());
        out.extend(0i32.to_le_bytes());
        std::fs::write(std::path::Path::new(&dir).join(format!("{name}.arrows")), out).unwrap();
    }

    /// Patch the record batch length (node 0 is moved aside temporarily so the
    /// only remaining i64 equal to `old` is the batch length).
    fn set_batch_length(data: &mut FlightData, old: i64, new: i64) {
        patch(data, Slot::NodeLength(0), 0x5a5a_5a5a_5a5a);
        let mut header = data.data_header.to_vec();
        let mut hits = vec![];
        for s in 0..header.len() - 7 {
            if header[s..s + 8] == old.to_le_bytes() {
                hits.push(s);
            }
        }
        assert_eq!(hits.len(), 1, "batch length occurrences");
        header[hits[0]..hits[0] + 8].copy_from_slice(&new.to_le_bytes());
        data.data_header = Bytes::from(header);
        patch(data, Slot::NodeLength(0), old);
    }

    fn run(name: &str, data: Vec<FlightData>) {
        dump(name, &data);
        let r = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
            let (mut o, mut e) = (Vec::new(), Vec::new());
            let r = session(true, ok(data), &mut o, &mut e);
            (r, String::from_utf8_lossy(&o).into_owned())
        }));
        match r {
            Ok((r, o)) => eprintln!("R3 {name}: result={r:?} stdout={o:?}"),
            Err(p) => eprintln!(
                "R3 {name}: PANIC {:?}",
                p.downcast_ref::<String>().cloned().or_else(|| p.downcast_ref::<&str>().map(|s| s.to_string()))
            ),
        }
    }

    fn fsl(size: i32, values: Vec<i8>) -> (Arc<Schema>, RecordBatch) {
        let column = FixedSizeListArray::new(
            Arc::new(Field::new_list_field(DataType::Int8, true)),
            size,
            Arc::new(Int8Array::from(values)),
            None,
        );
        let schema = Arc::new(Schema::new(vec![Field::new("f", column.data_type().clone(), true)]));
        let batch = RecordBatch::try_new(Arc::clone(&schema), vec![Arc::new(column)]).unwrap();
        (schema, batch)
    }

    fn union_schema(type_ids: Option<Vec<i32>>) -> Bytes {
        let mut fbb = flatbuffers::FlatBufferBuilder::new();
        let name = fbb.create_string("c0");
        let mut ib = arrow_ipc::IntBuilder::new(&mut fbb);
        ib.add_bitWidth(8);
        ib.add_is_signed(true);
        let int = ib.finish();
        let empty = fbb.create_vector::<flatbuffers::WIPOffset<arrow_ipc::Field>>(&[]);
        let mut b = arrow_ipc::FieldBuilder::new(&mut fbb);
        b.add_name(name);
        b.add_nullable(true);
        b.add_type_type(arrow_ipc::Type::Int);
        b.add_type_(int.as_union_value());
        b.add_children(empty);
        let kid = b.finish();
        let kids = fbb.create_vector(&[kid]);
        let ids = type_ids.map(|ids| fbb.create_vector(&ids));
        let mut u = arrow_ipc::UnionBuilder::new(&mut fbb);
        u.add_mode(arrow_ipc::UnionMode::Sparse);
        if let Some(ids) = ids {
            u.add_typeIds(ids);
        }
        let u = u.finish();
        let name = fbb.create_string("u");
        let mut b = arrow_ipc::FieldBuilder::new(&mut fbb);
        b.add_name(name);
        b.add_nullable(true);
        b.add_type_type(arrow_ipc::Type::Union);
        b.add_type_(u.as_union_value());
        b.add_children(kids);
        let field = b.finish();
        let fields = fbb.create_vector(&[field]);
        let mut s = arrow_ipc::SchemaBuilder::new(&mut fbb);
        s.add_fields(fields);
        let schema = s.finish();
        let mut m = arrow_ipc::MessageBuilder::new(&mut fbb);
        m.add_version(arrow_ipc::MetadataVersion::V5);
        m.add_header_type(MessageHeader::Schema);
        m.add_header(schema.as_union_value());
        let m = m.finish();
        fbb.finish(m, None);
        Bytes::copy_from_slice(fbb.finished_data())
    }
