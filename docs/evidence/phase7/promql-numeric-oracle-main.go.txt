package main

import (
	"fmt"
	"math"
	"strconv"
	"time"
)

//go:noinline
func toInt64(f float64) int64 { return int64(f) }

func formatTS(v float64, utc bool) string {
	sec, frac := math.Modf(v)
	t := time.Unix(toInt64(sec), toInt64(frac*1e9))
	if utc {
		return t.UTC().Format(time.RFC3339)
	}
	return t.Local().Format("2006-01-02T15:04:05-07:00")
}

func main() {
	for _, s := range []string{
		"0x1.8p-1075", "0x1p-1074", "0x1p-1075", "0x1p-1076", "-0x1p-1075",
		"0x0.0000000000001p-1022", "0x1p-1022",
		"0x1.fffffffffffffp1023", "0x1.fffffffffffff8p1023", "0x1p1024", "-0x1p1024",
		"0x1.00000000000008p0", "0x1.00000000000018p0", "0x1.0000000000000801p0",
		"0x000.8p1", "0X1P-1", "0x1_0p0", "0x1p1_0", "0x.p0", "0x1p", "0x1p+",
		"0x", "0x1.2.3p0", "0x1pz", "0x0p0", "0x1p-99999", "0x_1p4", "0x1",
	} {
		f, err := strconv.ParseFloat(s, 64)
		fmt.Printf("parse %q bits=%#016x err=%v\n", s, math.Float64bits(f), err != nil)
	}
	for _, v := range []float64{math.Inf(1), math.NaN(), 1e30, 9223372036854775808.0, 1.9, -1.9} {
		fmt.Printf("int64(%v)=%d round=%d\n", v, toInt64(v), int(math.Round(v)))
	}
	for _, v := range []float64{1704067200, 1704067200.75, -1.5, 0, 253402300800, 1e30} {
		fmt.Printf("ts(%v) utc=%s local=%s\n", v, formatTS(v, true), formatTS(v, false))
	}
	for _, v := range []float64{math.Inf(1) / 1e6, math.Inf(-1) / 1e6, math.NaN(), -0.04} {
		fmt.Printf("fixed1(%v)=%s\n", v, fmt.Sprintf("%.1f", v))
	}
	for _, sec := range []int64{-62135596801, -62198755200} {
		fmt.Printf("unix(%d)=%s\n", sec, time.Unix(sec, 0).UTC().Format(time.RFC3339))
	}
}
