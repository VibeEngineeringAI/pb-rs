fn unhex(s:&str)->Vec<u8>{s.as_bytes().chunks_exact(2).map(|b|u8::from_str_radix(std::str::from_utf8(b).unwrap(),16).unwrap()).collect()}
fn hex(s:&[u8])->String{s.iter().map(|b|format!("{b:02x}")).collect()}
fn main(){

for line in io::stdin().lock().lines(){let line=line.unwrap();let f:Vec<_>=line.split('\t').collect();let s=unhex(f[1]);let n:usize=f[2].parse().unwrap();let tail=unhex(f[3]);let bp=unhex(f[4]);
