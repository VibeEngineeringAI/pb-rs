func canonical(r rune) rune {
    smallest := r
    for next := unicode.SimpleFold(r); next != r; next = unicode.SimpleFold(next) {
        if next < smallest { smallest = next }
    }
    return smallest
}

}

pub(super) fn fold(c: char) -> u32 {
    if c.is_ascii() {
        return c.to_ascii_uppercase() as u32;
    }
    let rune = c as u32;
    FOLD.binary_search_by_key(&rune, |&(source, _)| source)
        .map_or(rune, |index| FOLD[index].1)
}

fn in_category(c: char, ranges: &[(u32, u32, u32)]) -> bool {
    let rune = c as u32;
    let index = ranges.partition_point(|&(lo, _, _)| lo <= rune);
    index.checked_sub(1).is_some_and(|index| {
        let (lo, hi, stride) = ranges[index];
        rune <= hi && (rune - lo).is_multiple_of(stride)
    })
}

    if c.is_ascii() {
