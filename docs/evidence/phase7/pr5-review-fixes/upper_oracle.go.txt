package main
import("fmt";"strings";"unicode")
func main() {
    fmt.Printf("// Go Unicode %s\n",unicode.Version)
    for c:=rune(0);c<=unicode.MaxRune;c++ {
        if c>=0xd800&&c<=0xdfff {continue}
        mapped:=unicode.ToUpper(c)
        if strings.ToUpper(string(c))!=string(mapped) {panic("rune/string uppercase differs")}
        if mapped!=c {fmt.Printf("%X %X\n",c,mapped)}
    }
}
